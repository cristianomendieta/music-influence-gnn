\section{Introdução}
%=====================================================

\begin{frame}{Introdução}
  \vfill
  \begin{description}[O que este exame apresenta]
    \setlength{\itemsep}{1.1em}
    \item[\azul{O tema}] Como conteúdos culturais ganham e perdem visibilidade
      em plataformas digitais.
    \item[\azul{O objeto empírico}] Música nos charts brasileiros do Spotify, 2017--2021.
    \item[\azul{A proposta}] Trocar a curva de popularidade isolada por um
      \dest{grafo heterogêneo temporal} de músicas, artistas e gêneros.
    \item[\azul{Este exame}] Proposta, resultados parciais e o que falta até a defesa.
  \end{description}
  \vfill
  \vfill
\end{frame}

    \node[font=\small,ufprC] at (5.4,-0.2) {funk};
  \end{tikzpicture}
  \end{center}
  \vspace{0.5em}
  \begin{block}{Definição}
    Entidades \dest{semelhantes} tendem a se \dest{conectar}.
  \end{block}
  \begin{itemize}
    \item É a \dest{premissa} que valida a agregação por vizinhança
    \item Sem ela, a média dos vizinhos \emph{apaga} o sinal do nó
    \item Evidência no nosso grafo: \emph{clustering} \azul{0,51} entre músicas que sobem juntas
          nas paradas e \azul{0,60} entre gêneros que coocorrem, cerca de \dest{17$\times$} o
          esperado ao acaso
  \end{itemize}
\end{frame}

  \begin{block}{Por que desacoplar}
    O codificador não depende da música-alvo: \dest{uma passada por semana}, reutilizada por todas.\\
    $\sim$500 mil passagens $\rightarrow$ \dest{260}.
  \end{block}
\end{frame}

%-----------------------------------------------------
\begin{frame}{Codificador --- a mensagem}
  \vfill
  \[
  \mathbf{h}_v^{(r,\ell)} \;=\;
  \underbrace{\mathbf{W}^{(\ell)}_{r,\text{self}}\,\mathbf{h}_v^{(\ell-1)}}_{\text{\footnotesize\textcolor{ufprA}{o próprio } } v}
  \;+\;
  \underbrace{\mathbf{W}^{(\ell)}_{r,\text{viz}} \cdot \operatorname{mean}_{u \in \mathcal{N}_r(v)} \mathbf{h}_u^{(\ell-1)}}_{\text{\footnotesize\textcolor{ufprB}{os vizinhos pela relação } } r}
  \]
  \vfill
  \begin{itemize}
    \item $\mathcal{N}_r(v)$ --- vizinhos \dest{só por aquela relação}
    \item $\mathbf{W}_{r}$ --- matrizes \dest{diferentes por relação}: é o que torna o modelo heterogêneo
    \item $\operatorname{mean}$ --- normaliza pelo grau
    \item $\mathbf{h}_v^{(\ell-1)}$ --- o vetor do nó $v$ ao sair da camada
      anterior; na primeira camada, são os \dest{atributos de entrada}
  \end{itemize}
  \vfill
\end{frame}

%-----------------------------------------------------
\begin{frame}{Codificador --- combinando relações}
  \vfill
  \[
  \mathbf{h}_v^{(\ell)} = \operatorname{ReLU}\!\Bigg( \sum_{r \,\in\, \mathcal{R}(v)} \mathbf{h}_v^{(r,\ell)} \Bigg)
  \]
  \vfill
  \begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \begin{itemize}
      \item Soma $\Rightarrow$ peso \dest{uniforme} por relação
      \item É aqui que o \azul{HGT} entraria
    \end{itemize}
  \end{column}
  \begin{column}{0.5\textwidth}
    \begin{block}{Por que $L \geq 2$}
      \begin{center}
      \begin{tikzpicture}[every node/.style={font=\scriptsize},node distance=1.1cm]
        \node[draw=ufprC,fill=ufprC!15,rounded corners,thick] (g) {genre};

        \draw[-{Latex[length=2mm]},thick] (a) -- node[above]{\tiny 2} (m);
      \end{tikzpicture}
      \end{center}
      \small O gênero só alcança a música em \dest{dois passos}.
    \end{block}
  \end{column}
  \end{columns}
  \vfill
\end{frame}


%-----------------------------------------------------
\begin{frame}{Cabeça temporal}
  \vfill
  \[ \mathbf{s}_i = \big( \mathbf{z}_i^{(w-W)}, \dots, \mathbf{z}_i^{(w-1)} \big)
     \qquad\longrightarrow\qquad
     \hat{y}_i = \operatorname{clamp}\big( y_i^{(w-1)} + \Delta_i,\; 0,\; 0{,}5 \big),
     \quad \Delta_i = \operatorname{MLP}(\mathbf{h}_i) \]
  \vfill
  \begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \begin{itemize}
      \item Janela \dest{causal}: nunca inclui $w$; a âncora $y^{(w-1)}$ é sempre passada
      \item Zeros $+$ \azul{máscara} antes da estreia
      \item GRU de uma camada $+$ MLP
    \end{itemize}
  \end{column}
  \begin{column}{0.5\textwidth}
    \begin{block}{Cabeça residual}
      A rede prevê o \dest{incremento} sobre o último valor observado, não o valor
      absoluto.\\[0.4em]
      Última camada com \azul{inicialização em zero}: o modelo não treinado
      \emph{reproduz a persistência}.
    \end{block}
  \end{column}
  \end{columns}
  \vfill
  \begin{center}
  \small O codificador diz \azul{``com quem''}; a GRU diz \azul{``como vem mudando''}.
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Quatro frentes até a defesa}
  \vfill
  \begin{enumerate}
    \item \azul{\emph{Baseline} neural sem grafo} --- GRU sobre a série isolada.\\
      {\small O \dest{grupo de controle}: separa flexibilidade neural de estrutura relacional.}
    \item \azul{Interpretabilidade e ablação} --- remover cada tipo de aresta; análogos de
      $\beta$, $\gamma$, $R_0$.\\
      {\small Responde \dest{Q2} e \dest{Q3}.}
    \item \azul{Arquitetura alternativa} --- HGT ou atenção por relação, tudo o mais fixo.\\
      {\small Testa robustez e dá interpretabilidade extra.}
    \item \azul{Consolidação e redação} --- comparação única sob as mesmas métricas.
  \end{enumerate}
  \vfill
\end{frame}
